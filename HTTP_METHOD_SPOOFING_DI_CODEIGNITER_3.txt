 * Implementasi untuk mensimulasikan HTTP methods PUT, PATCH, DELETE

// ==========================================
// 1. CORE HOOKS - application/hooks/Http_method_spoofing.php
// ==========================================
class Http_method_spoofing {
    
    public function override_method() {
        $CI =& get_instance();
        
        // Cek apakah ada _method di POST data
        $method = $CI->input->post('_method');
        
        if ($method) {
            $method = strtoupper($method);
            
            // Validasi method yang diizinkan
            $allowed_methods = ['PUT', 'PATCH', 'DELETE', 'POST', 'GET'];
            
            if (in_array($method, $allowed_methods)) {
                // Override $_SERVER['REQUEST_METHOD']
                $_SERVER['REQUEST_METHOD'] = $method;
                $_SERVER['ORIGINAL_REQUEST_METHOD'] = $_SERVER['REQUEST_METHOD'] ?? 'POST';
            }
        }
        
        // Cek header X-HTTP-Method-Override (untuk AJAX)
        $header_method = $CI->input->get_request_header('X-HTTP-Method-Override', TRUE);
        if ($header_method) {
            $header_method = strtoupper($header_method);
            $allowed_methods = ['PUT', 'PATCH', 'DELETE', 'POST', 'GET'];
            
            if (in_array($header_method, $allowed_methods)) {
                $_SERVER['REQUEST_METHOD'] = $header_method;
                $_SERVER['ORIGINAL_REQUEST_METHOD'] = $_SERVER['REQUEST_METHOD'] ?? 'POST';
            }
        }
    }
}

// 2. HELPER FUNCTION - application/helpers/http_helper.php
// ==========================================
if (!function_exists('method_field')) {
    /**
     * Generate hidden input untuk method spoofing
     */
    function method_field($method) {
        return '<input type="hidden" name="_method" value="' . strtoupper($method) . '">';
    }
}

if (!function_exists('is_method')) {
    /**
     * Cek HTTP method saat ini
     */
    function is_method($method) {
        return strtoupper($_SERVER['REQUEST_METHOD']) === strtoupper($method);
    }
}

if (!function_exists('current_method')) {
    /**
     * Get current HTTP method
     */
    function current_method() {
        return $_SERVER['REQUEST_METHOD'];
    }
}

// 3. EXTENDED INPUT CLASS - application/core/MY_Input.php
// ==========================================
class MY_Input extends CI_Input {
    
    /**
     * Get HTTP method
     */
    public function method($upper = TRUE) {
        $method = $_SERVER['REQUEST_METHOD'];
        return $upper ? strtoupper($method) : strtolower($method);
    }
    
    /**
     * Check if method is specific type
     */
    public function is_method($method) {
        return $this->method() === strtoupper($method);
    }
    
    /**
     * Get original method before spoofing
     */
    public function original_method() {
        return $_SERVER['ORIGINAL_REQUEST_METHOD'] ?? $_SERVER['REQUEST_METHOD'];
    }
    
    /**
     * Get PUT data
     */
    public function put($index = NULL, $xss_clean = NULL) {
        return $this->_fetch_from_array($_POST, $index, $xss_clean);
    }
    
    /**
     * Get PATCH data
     */
    public function patch($index = NULL, $xss_clean = NULL) {
        return $this->_fetch_from_array($_POST, $index, $xss_clean);
    }
    
    /**
     * Get DELETE data
     */
    public function delete($index = NULL, $xss_clean = NULL) {
        return $this->_fetch_from_array($_POST, $index, $xss_clean);
    }
}

// 4. CONTROLLER EXAMPLE - application/controllers/Users.php
// ==========================================
class Users extends CI_Controller {
    
    public function __construct() {
        parent::__construct();
        $this->load->helper(['url', 'form', 'http']);
        $this->load->library('form_validation');
    }
    
    /**
     * GET /users - List all users
     */
    public function index() {
        if (!$this->input->is_method('GET')) {
            show_404();
        }
        
        // Logic untuk menampilkan daftar users
        $data['users'] = $this->user_model->get_all();
        $this->load->view('users/index', $data);
    }
    
    /**
     * GET /users/create - Show create form
     */
    public function create() {
        if (!$this->input->is_method('GET')) {
            show_404();
        }
        
        $this->load->view('users/create');
    }
    
    /**
     * POST /users - Store new user
     */
    public function store() {
        if (!$this->input->is_method('POST')) {
            show_404();
        }
        
        $this->form_validation->set_rules('name', 'Name', 'required');
        $this->form_validation->set_rules('email', 'Email', 'required|valid_email');
        
        if ($this->form_validation->run() === FALSE) {
            $this->load->view('users/create');
        } else {
            $data = [
                'name' => $this->input->post('name'),
                'email' => $this->input->post('email')
            ];
            
            $this->user_model->create($data);
            redirect('users');
        }
    }
    
    /**
     * GET /users/{id} - Show specific user
     */
    public function show($id) {
        if (!$this->input->is_method('GET')) {
            show_404();
        }
        
        $data['user'] = $this->user_model->get($id);
        $this->load->view('users/show', $data);
    }
    
    /**
     * GET /users/{id}/edit - Show edit form
     */
    public function edit($id) {
        if (!$this->input->is_method('GET')) {
            show_404();
        }
        
        $data['user'] = $this->user_model->get($id);
        $this->load->view('users/edit', $data);
    }
    
    /**
     * PUT /users/{id} - Update user
     */
    public function update($id) {
        if (!$this->input->is_method('PUT')) {
            show_404();
        }
        
        $this->form_validation->set_rules('name', 'Name', 'required');
        $this->form_validation->set_rules('email', 'Email', 'required|valid_email');
        
        if ($this->form_validation->run() === FALSE) {
            $data['user'] = $this->user_model->get($id);
            $this->load->view('users/edit', $data);
        } else {
            $data = [
                'name' => $this->input->put('name'),
                'email' => $this->input->put('email')
            ];
            
            $this->user_model->update($id, $data);
            redirect('users');
        }
    }
    
    /**
     * DELETE /users/{id} - Delete user
     */
    public function destroy($id) {
        if (!$this->input->is_method('DELETE')) {
            show_404();
        }
        
        $this->user_model->delete($id);
        
        // Untuk AJAX request
        if ($this->input->is_ajax_request()) {
            $this->output
                ->set_content_type('application/json')
                ->set_output(json_encode(['status' => 'success']));
        } else {
            redirect('users');
        }
    }
    
    /**
     * Handle multiple methods in one function
     */
    public function user_action($id = null) {
        switch ($this->input->method()) {
            case 'GET':
                if ($id) {
                    return $this->show($id);
                } else {
                    return $this->index();
                }
                break;
                
            case 'POST':
                return $this->store();
                break;
                
            case 'PUT':
                return $this->update($id);
                break;
                
            case 'DELETE':
                return $this->destroy($id);
                break;
                
            default:
                show_404();
        }
    }
}

                type: 'POST',
                headers: {
                    'X-HTTP-Method-Override': 'DELETE'

                        location.reload();
